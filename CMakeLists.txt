cmake_minimum_required(VERSION 3.22)

# Static MSVC runtime (/MT, /MTd) for every target, JUCE included: VstPlayer.exe then runs on a
# clean Windows 10/11 without the Visual C++ Redistributable (see tools/package.ps1).
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

project(VstPlayer VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT MSVC)
    message(FATAL_ERROR "vst-player requires MSVC (Visual Studio 2022 Build Tools). Run tools/build.ps1.")
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(juce)
include(httplib)
include(msvc-flags)

juce_add_gui_app(VstPlayer
    PRODUCT_NAME "VstPlayer"
    COMPANY_NAME "Jape"
    VERSION ${PROJECT_VERSION})

target_sources(VstPlayer PRIVATE
    src/Main.cpp
    src/app/MainWindow.cpp
    src/app/AppSettings.cpp
    src/app/CommandLine.cpp
    src/app/InstanceSelfTest.cpp
    src/app/MidiSelfTest.cpp
    src/app/OfflineRender.cpp
    src/app/ProjectSelfTest.cpp
    src/app/ProcessTools.cpp
    src/app/Translations.cpp
    src/app/UiSelfTest.cpp
    src/api/ApiController.cpp
    src/api/ApiServer.cpp
    src/engine/AudioEngine.cpp
    src/engine/InstanceGraph.cpp
    src/engine/MidiRouter.cpp
    src/engine/PluginManager.cpp
    src/project/Project.cpp
    src/project/ProjectController.cpp
    src/project/ProjectSerializer.cpp
    src/project/RecentProjects.cpp
    src/ui/AddInstanceDialog.cpp
    src/ui/InstancePanel.cpp
    src/ui/MainContent.cpp
    src/ui/MidiPortsWindow.cpp
    src/ui/PluginEditorWindow.cpp
    src/ui/PluginListWindow.cpp
    src/ui/Theme.cpp)

target_include_directories(VstPlayer PRIVATE src)

# examples/carrilion-client.js embedded in the exe: the API serves it at GET /client.js, so a page
# can do `await import('http://127.0.0.1:9471/client.js')`. Regenerated when the file changes.
set(VSTPLAYER_CLIENT_JS "${CMAKE_CURRENT_SOURCE_DIR}/examples/carrilion-client.js")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${VSTPLAYER_CLIENT_JS}")
file(READ "${VSTPLAYER_CLIENT_JS}" _client_hex HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _client_bytes "${_client_hex}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/ClientScript.h.tmp"
    "// Generated by CMakeLists.txt from examples/carrilion-client.js. Do not edit.\n#pragma once\n"
    "static const unsigned char kClientScript[] = { ${_client_bytes} };\n")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/generated/ClientScript.h.tmp"
               "${CMAKE_CURRENT_BINARY_DIR}/generated/ClientScript.h" COPYONLY)
target_include_directories(VstPlayer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

target_compile_definitions(VstPlayer PRIVATE
    JUCE_PLUGINHOST_VST3=1
    JUCE_PLUGINHOST_VST=0
    JUCE_PLUGINHOST_AU=0
    JUCE_PLUGINHOST_LADSPA=0
    JUCE_PLUGINHOST_LV2=0
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
    JUCE_REPORT_APP_USAGE=0
    JUCE_APPLICATION_NAME_STRING="$<TARGET_PROPERTY:VstPlayer,JUCE_PRODUCT_NAME>"
    JUCE_APPLICATION_VERSION_STRING="$<TARGET_PROPERTY:VstPlayer,JUCE_VERSION>")

# Optional ASIO support: only if the Steinberg ASIO SDK was dropped into third_party/asiosdk.
set(VSTPLAYER_ASIOSDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/asiosdk")
if(EXISTS "${VSTPLAYER_ASIOSDK_DIR}/common/iasiodrv.h")
    message(STATUS "ASIO SDK found: enabling JUCE_ASIO")
    target_compile_definitions(VstPlayer PRIVATE JUCE_ASIO=1)
    target_include_directories(VstPlayer PRIVATE "${VSTPLAYER_ASIOSDK_DIR}/common")
else()
    message(STATUS "ASIO SDK not found in third_party/asiosdk: ASIO disabled (WASAPI only)")
endif()

target_link_libraries(VstPlayer
    PRIVATE
        juce::juce_gui_extra
        juce::juce_audio_utils
        juce::juce_audio_processors
        juce::juce_audio_devices
        httplib::httplib
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags)

vstplayer_set_msvc_flags(VstPlayer)

# Tests (ctest): round-trip of the project model / JSON serializer. juce_core only.
enable_testing()

juce_add_console_app(ProjectRoundTripTest PRODUCT_NAME "ProjectRoundTripTest")

target_sources(ProjectRoundTripTest PRIVATE
    tests/ProjectRoundTripTest.cpp
    src/project/Project.cpp
    src/project/ProjectSerializer.cpp)

target_include_directories(ProjectRoundTripTest PRIVATE src)

target_compile_definitions(ProjectRoundTripTest PRIVATE
    JUCE_USE_CURL=0
    JUCE_WEB_BROWSER=0
    JUCE_STANDALONE_APPLICATION=1
    VSTPLAYER_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")

target_link_libraries(ProjectRoundTripTest
    PRIVATE
        juce::juce_core
    PUBLIC
        juce::juce_recommended_config_flags)

vstplayer_set_msvc_flags(ProjectRoundTripTest)

add_test(NAME project_roundtrip COMMAND ProjectRoundTripTest)
